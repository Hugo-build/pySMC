┌─────────────────────────────────────────────────────────────────────────┐

┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
┃  LAYER 1: Base Interface (Universal)                                   ┃
┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    ┌───────────────────────┐
    │   BaseSurrogate       │  ← Abstract interface for ALL models
    │      (ABC)            │
    ├───────────────────────┤
    │ + fit(X, y)          │
    │ + predict(X_star)    │
    │ + is_fitted()        │
    │ + validate_inputs()  │
    └───────────────────────┘

┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓

┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛

    ├───────────────────────────────┤
    │ + suggest_next_sample()       │  ← Intelligent sampling
    │ + add_sample()                │  ← Incremental updates
    │ + get_training_data()         │
    └───────────────────────────────┘

┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓

KEY BENEFITS:

  ✅ GP.py stays INDEPENDENT  → Can use it standalone
  ✅ ADAPTER pattern          → Easy to add new models
  ✅ ADAPTIVE methods         → Built-in intelligent sampling
  ✅ MODULAR design           → Mix and match components
  ✅ JAX-powered              → Fast, differentiable
  ✅ EXTENSIBLE               → Add features without breaking code

USAGE EXAMPLES:

1. Basic GP Surrogate:
   surrogate = GaussianProcessAdapter.create(kernel_type='matern52')
   surrogate = surrogate.fit(X, y, optimize_hyperparameters=True)
   mean, std = surrogate.predict(X_test)

2. Adaptive Kriging:
   agp = AdaptiveGaussianProcess.create(kernel_type='rbf')
   agp = agp.fit(X_init, y_init)
   
   for i in range(n_iterations):
       X_next = agp.suggest_next_sample(X_candidates, 
                                        acquisition=VarianceReduction())
       y_next = expensive_function(X_next)
       agp = agp.add_sample(X_next, y_next, refit=True)

3. Bayesian Optimization:
   agp = AdaptiveGaussianProcess.create()
   agp = agp.fit(X_init, y_init)
   
   X_next = agp.suggest_next_sample(X_candidates,
                                    acquisition=ExpectedImprovement())
